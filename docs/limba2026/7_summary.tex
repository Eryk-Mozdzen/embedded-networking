\section{Podsumowanie}
    \begin{frame}{Zady i walety komunikacji sieciowej}
        Zalety:
        \begin{itemize}
            \item wsparcie na każdym systemie operacyjnym
            \item łatwa integracja z istniejącym oprogramowaniem
            \item umożliwia używanie dobrodziejstw sieciówki: routing, usługi, narzędzia
            \item \textbf{architektora fizyczna przestaje implikować architekturę logiczną}
        \end{itemize}
        Wady:
        \begin{itemize}
            \item wymaga dużo RAMu
            \item nie jest hard real-time (jeszcze, czekam na TSN)
            \item wymaga wiedzy z sieciówki
        \end{itemize}
    \end{frame}
